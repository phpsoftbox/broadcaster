# DI

## Пример с PHP-DI

```php
use DI\ContainerBuilder;
use PhpSoftBox\Broadcaster\Contracts\PushrRegistryBuilderInterface;
use PhpSoftBox\Broadcaster\Pushr\PushrAppRegistry;
use PhpSoftBox\Broadcaster\Pushr\PushrServer;
use PhpSoftBox\Broadcaster\Registry\CompositePushrRegistryBuilder;
use PhpSoftBox\Broadcaster\Registry\ConfigPushrRegistrySource;
use PhpSoftBox\Config\Config;

$builder = new ContainerBuilder();
$builder->addDefinitions([
    PushrRegistryBuilderInterface::class => function (Config $config): PushrRegistryBuilderInterface {
        return new CompositePushrRegistryBuilder([
            new ConfigPushrRegistrySource($config),
        ]);
    },
    PushrAppRegistry::class => function (PushrRegistryBuilderInterface $builder): PushrAppRegistry {
        return $builder->build();
    },
    PushrServer::class => function (PushrAppRegistry $registry): PushrServer {
        return new PushrServer($registry, host: '0.0.0.0', port: 8080);
    },
]);
```
